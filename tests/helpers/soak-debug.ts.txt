import { GameEngine } from '../src/core/engine';
import { Bot } from './helpers/bot';
test('debug soak', () => {
  const e = GameEngine.newGame(Number(process.env.SEED ?? 42));
  const bot = new Bot(e);
  bot.setup();
  const t0 = Date.now();
  const N = Number(process.env.TICKS ?? 1500);
  for (let i = 0; i < N; i++) {
    bot.step();
    e.tick();
    const s = e.state;
    const d = s.derived;
    if (i % 50 === 0)
      console.log(
        `t${s.tick} y${s.year} era${s.era} pop${d.population} $${Math.round(s.money)} net${d.netIncome.toFixed(1)} H${d.happiness.toFixed(0)} RCI ${d.demand.R.toFixed(0)}/${d.demand.C.toFixed(0)}/${d.demand.I.toFixed(0)} food${d.foodSatisfaction.toFixed(2)} unemp${d.unemployment.toFixed(2)} res${d.researchPerTick.toFixed(1)} cul${d.culturePerTick.toFixed(1)} exp${d.budget.exports.toFixed(0)} cap${d.exportCapacity.toFixed(0)} t${s.research.done.length}/c${s.civics.done.length} ${s.research.current}/${s.civics.current} b${s.buildings.length} pw${d.power.supply.toFixed(0)}/${d.power.demand.toFixed(0)} tax${d.budget.taxR.toFixed(0)}/${d.budget.taxC.toFixed(0)}/${d.budget.taxI.toFixed(0)} up${d.budget.upkeepBuildings.toFixed(0)}+${d.budget.upkeepTransport.toFixed(0)} deals${d.budget.deals.toFixed(0)} adm${d.budget.admin.toFixed(0)}`,
      );
  }
  console.log(bot.log.filter((l) => l.includes('ERA')).join('\n'));
  console.log('ms', Date.now() - t0);
});
